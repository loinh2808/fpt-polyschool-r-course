Outlier Detection by IQR

q1 <- quantile(adv_desc$outlier_score, 0.25, na.rm = TRUE)
q3 <- quantile(adv_desc$outlier_score, 0.75, na.rm = TRUE)
iqr_value <- IQR(adv_desc$outlier_score, na.rm = TRUE)

lower_bound <- q1 - 1.5 * iqr_value
upper_bound <- q3 + 1.5 * iqr_value

adv_desc |>
  filter(
    outlier_score < lower_bound |
      outlier_score > upper_bound
  )

7.3 Advanced Regression Models

NoteKnowledge Notes

What?

Advanced regression mở rộng hồi quy tuyến tính cho nhiều loại biến phụ thuộc.

When?

Outcome type Model
Binary Logistic regression
Ordinal Ordinal regression
Count Poisson regression
Overdispersed count Negative binomial
Continuous Linear regression

Import Data

adv_reg <- read_excel(
  advanced_file,
  sheet = "02_Advanced_Regression"
) |>
  clean_names()

Logistic Regression

logit_model <- glm(
  adopted_ai ~ digital_readiness + leadership_support + workload + training_hours,
  data = adv_reg,
  family = binomial
)

tidy(logit_model, exponentiate = TRUE, conf.int = TRUE)

Ordinal Regression

if (requireNamespace("MASS", quietly = TRUE)) {
  ordinal_model <- MASS::polr(
    as.factor(ordinal_performance) ~ digital_readiness + leadership_support + motivation,
    data = adv_reg,
    Hess = TRUE
  )

  broom::tidy(ordinal_model, conf.int = TRUE)
}

Poisson Regression

poisson_model <- glm(
  absences_count ~ workload + job_satisfaction + training_hours,
  data = adv_reg,
  family = poisson
)

tidy(poisson_model, exponentiate = TRUE, conf.int = TRUE)

Mediation Logic

med_model_a <- lm(
  mediator_engagement ~ leadership_support,
  data = adv_reg
)

med_model_b <- lm(
  productivity_score ~ leadership_support + mediator_engagement,
  data = adv_reg
)

tidy(med_model_a)
tidy(med_model_b)

Moderation Logic

moderation_model <- lm(
  productivity_score ~ digital_readiness * moderator_autonomy,
  data = adv_reg
)

tidy(moderation_model)
WarningCommon Mistakes

Không chọn mô hình chỉ vì quen dùng lm().

Cần chọn mô hình theo loại biến phụ thuộc.
